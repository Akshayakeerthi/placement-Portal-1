Table users {
  id int [pk, increment]
  name varchar(120) [not null]
  email varchar(255) [not null, unique]
  password_hash varchar(255) [not null]
  role varchar(20) [not null, note: "ADMIN | COMPANY | STUDENT"]
  is_active boolean [not null, default: true]
  is_blacklisted boolean [not null, default: false]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    email [unique]
    role
  }
}

Table company_profiles {
  id int [pk, increment]
  user_id int [not null, unique]
  company_name varchar(150) [not null]
  website varchar(255)
  description text
  approved boolean [not null, default: false]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    user_id [unique]
  }
}

Table student_profiles {
  id int [pk, increment]
  user_id int [not null, unique]
  branch varchar(100) [not null]
  graduation_year int [not null]
  cgpa float [not null]
  resume_path varchar(255)
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    user_id [unique]
  }
}

Table placement_drives {
  id int [pk, increment]
  company_id int [not null]
  title varchar(150) [not null]
  description text [not null]
  eligible_branches varchar(300) [not null, note: "Comma-separated branches"]
  min_cgpa float [not null]
  graduation_year int [not null]
  deadline datetime [not null]
  approved boolean [not null, default: false]
  closed boolean [not null, default: false]
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table applications {
  id int [pk, increment]
  student_id int [not null]
  drive_id int [not null]
  status varchar(50) [not null, default: 'APPLIED', note: "APPLIED | SHORTLISTED | INTERVIEW_SCHEDULED | SELECTED | REJECTED"]
  interview_schedule varchar(255)
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (student_id, drive_id) [unique, name: 'uq_student_drive_application']
  }
}
